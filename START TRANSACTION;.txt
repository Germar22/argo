START TRANSACTION;

-- 1. Delete tax credit references
DELETE tci
FROM tax_credit_invoices tci
INNER JOIN invoice i ON i.ID = tci.INVOICE_ID
WHERE i.`DATE` >= '2026-01-01'
  AND i.`DATE` < '2026-08-01';

-- 2. Delete invoice details
DELETE ii
FROM invoice_items ii
INNER JOIN invoice i ON i.ID = ii.INVOICE_ID
WHERE i.`DATE` >= '2026-01-01'
  AND i.`DATE` < '2026-08-01';

-- 3. Delete invoice headers
DELETE FROM invoice
WHERE `DATE` >= '2026-01-01'
  AND `DATE` < '2026-08-01';

-- 4. Confirm remaining invoices in the target period
SELECT COUNT(*) AS RemainingInvoices
FROM invoice
WHERE `DATE` >= '2026-01-01'
  AND `DATE` < '2026-08-01';

-- 5. Confirm remaining invoice items in the target period
SELECT COUNT(*) AS RemainingInvoiceItems
FROM invoice_items ii
INNER JOIN invoice i ON i.ID = ii.INVOICE_ID
WHERE i.`DATE` >= '2026-01-01'
  AND i.`DATE` < '2026-08-01';
COMMIT;
